pragma solidity ^0.8.29;

import "@openzeppelin/contracts-upgradeable/token/ERC1155/ERC1155Upgradeable.sol";  // ✅ Aggiunto
import "@openzeppelin/contracts-upgradeable/token/ERC1155/extensions/ERC1155SupplyUpgradeable.sol";
import "@openzeppelin/contracts-upgradeable/access/OwnableUpgradeable.sol";
import "@openzeppelin/contracts-upgradeable/proxy/utils/UUPSUpgradeable.sol";
import "../core_justice/RefundManager.sol";

contract SolidaryComics_StellaDoppia is
    Initializable,
    ERC1155Upgradeable,  // ✅ Aggiunto
    ERC1155SupplyUpgradeable,
    OwnableUpgradeable,
    UUPSUpgradeable,
    RefundManager
{
    // ... (resto del codice invariato)

    function _update(
        address from,
        address to,
        uint256[] memory ids,
        uint256[] memory values
    ) internal virtual override {
        if (from != address(0) && to != address(0) && feePercent > 0) {
            for (uint256 i = 0; i < ids.length; ++i) {
                if (ids[i] == FT_COMIX_ID && values[i] > 0) {
                    uint256 fee = (values[i] * feePercent) / 100;
                    if (fee > 0) {
                        fee = fee > values[i] ? values[i] : fee;
                        values[i] -= fee;

                        uint256[] memory feeIds = new uint256[](1);
                        uint256[] memory feeValues = new uint256[](1);
                        feeIds[0] = ids[i];
                        feeValues[0] = fee;

                        super._update(from, charityWallet, feeIds, feeValues);
                        emit ComicDonation(from, fee);
                    }
                }
            }
        }
        super._update(from, to, ids, values);  // ✅ Trasferimento principale
    }
}
