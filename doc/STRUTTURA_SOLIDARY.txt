pragma solidity ^0.8.0;

contract SolidaryMintableToken {
    uint256 public mintPrice = 0.01 ether; // prezzo uniforme per mint
    address public nftContract; // indirizzo del contratto NFT

    event Minted(address indexed user, uint256 ftAmount, uint256 nftAmount);

    function mint() external payable {
        require(msg.value == mintPrice, "Invalid price");

        uint256 ftAmount = (msg.value * 45) / 100; // 45% FT
        uint256 nftAmount = msg.value - ftAmount;  // 55% NFT

        // Mint FT (es. ERC20)
        _mintFT(msg.sender, ftAmount);

        // Mint NFT (es. chiama funzione mint del contratto NFT)
        INFT(nftContract).mintNFT{value: nftAmount}(msg.sender);

        emit Minted(msg.sender, ftAmount, nftAmount);
    }

    function _mintFT(address to, uint256 amount) internal {
        // Implementa la logica ERC20 mint qui
    }
}

interface INFT {
    function mintNFT(address to) external payable;
}
